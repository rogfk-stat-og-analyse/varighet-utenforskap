import fdb/ARBSTATUS_PERS_UTENFOR 2019-11-16 as ARBSTATUS2019
tabulate ARBSTATUS2019, missing

// kopi av variabel
generate Status2019 = "5 - Ukjent"
replace Status2019 = "1 - Innenfor" if ARBSTATUS2019 == '0'
replace Status2019 = "2 - Utenfor" if ARBSTATUS2019 == '1'

tabulate Status2019

//henter Prioritert arbeidsstyrkestatus
import fdb/ARBSTATUS_PERS_STATUS8 2019-11-16 as ARBSTATUS2019_pri
tabulate ARBSTATUS2019_pri, missing

tabulate ARBSTATUS2019_pri Status2019 

//Lager en kategorisering basert på prioritert arbeidsstatus
generate Status4siffer2019 = substr(ARBSTATUS2019_pri,1,4)
tabulate Status4siffer2019

generate StatusHovedkat2019 = ""

// Arbeid (sterk)
// Omfatter personer der hovedstatus er arbeid og der de fire siste sifrene er 0000, altså ingen sekundærstatus i denne variabelen.
// Dette tolkes som den sterkeste formen for tilknytning til arbeid.
replace StatusHovedkat2019 = "Arbeid (sterk)" if (Status4siffer2019 == "0808" | Status4siffer2019 == "0816" | Status4siffer2019 == "0832" | Status4siffer2019 == "1608") & substr(ARBSTATUS2019_pri,5,4) == "0000"

// Arbeid (kombinert)
// Omfatter personer der hovedstatus er arbeid, men som samtidig har en sekundærstatus.
// Dette regnes fortsatt som arbeidstilknytning, men en svakere eller mer sammensatt form enn "Arbeid (sterk)".
replace StatusHovedkat2019 = "Arbeid (kombinert)" if (Status4siffer2019 == "0808" | Status4siffer2019 == "0816" | Status4siffer2019 == "0832" | Status4siffer2019 == "1608") & substr(ARBSTATUS2019_pri,5,4) != "0000"

// Utdanning
// Omfatter personer med ordinær utdanning som høyest prioriterte status.
// Dette regnes som innenfor arbeid, utdanning og opplæring, selv om personen ikke er i arbeid.
replace StatusHovedkat2019 = "Utdanning" if Status4siffer2019 == "4008"

// Tiltak/kvalifisering
// Omfatter ordinære tiltak, individstønad/tiltakspenger og introduksjonsprogram.
// Disse regnes som innenfor fordi personen deltar i kvalifiserende eller arbeidsrettet aktivitet, men tilknytningen er svakere enn arbeid og ordinær utdanning.
replace StatusHovedkat2019 = "Tiltak/kvalifisering" if Status4siffer2019 == "3208" | Status4siffer2019 == "3216" | Status4siffer2019 == "3240"

// Arbeidsledig
// Omfatter registrert helt arbeidsledig og dagpengemottakere.
// Disse regnes som utenfor i denne inndelingen, men som en gruppe med relativ nærhet til arbeid sammenlignet med flere andre utenfor-grupper.
replace StatusHovedkat2019 = "Arbeidsledig" if Status4siffer2019 == "2408" | Status4siffer2019 == "3224"

// Helse (midlertidig)
// Omfatter sykepenger, nedsatt arbeidsevne på tiltak, AAP og nedsatt arbeidsevne ikke på tiltak.
// Dette er en analytisk samlegruppe for helserelatert eller arbeidsevnerelatert utenforskap.
// Kategorien er grov: særlig 4816 og 6432 kan i praksis være langvarige tilstander, men de holdes her samlet for å få en håndterbar hovedinndeling.
replace StatusHovedkat2019 = "Helse (midlertidig)" if Status4siffer2019 == "4808" | Status4siffer2019 == "4816" | Status4siffer2019 == "4848" | Status4siffer2019 == "6432"

// Uføre (varig)
// Omfatter varig uførepensjon/uføretrygd.
// Dette skilles ut som egen kategori fordi det normalt representerer en mer varig og avklart form for utenforskap enn de øvrige helserelaterte statusene.
replace StatusHovedkat2019 = "Uføre (varig)" if Status4siffer2019 == "4864"

// Sosiale ytelser
// Omfatter kontantstøtte, overgangsstønad og sosialhjelp.
// Dette er en samlegruppe for personer uten arbeid, utdanning eller tiltak som hovedstatus, og med svak eller indirekte tilknytning til arbeidsmarkedet.
replace StatusHovedkat2019 = "Sosiale ytelser" if Status4siffer2019 == "6408" | Status4siffer2019 == "6416" | Status4siffer2019 == "6424"

// Pensjon
// Omfatter etterlattepensjon, AFP, supplerende stønad, alderspensjon og andre pensjoner.
// Kategorien holdes separat fordi dette i mange tilfeller ikke bør forstås som "utenforskap" i vanlig sosialpolitisk forstand, men som livsfase eller pensjonsstatus.
replace StatusHovedkat2019 = "Pensjon" if Status4siffer2019 == "5608" | Status4siffer2019 == "5616" | Status4siffer2019 == "5624" | Status4siffer2019 == "5632" | Status4siffer2019 == "5640"

// Ukjent
// Omfatter personer med ukjent status i datagrunnlaget.
// Holdes som egen kategori for å synliggjøre datamessig usikkerhet og unngå at disse feilaktig plasseres i andre grupper.
replace StatusHovedkat2019 = "Ukjent" if Status4siffer2019 == "7208"

tabulate StatusHovedkat2019, freq cellpct missing

tabulate ARBSTATUS2019_pri StatusHovedkat2019, freq cellpct missing
tabulate StatusHovedkat2019 Status2019, freq colpct cellpct missing
